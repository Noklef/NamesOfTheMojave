; Loads every *.txt file in Plugins/ and Rules/, and the name pools that the rules use.
; Add-ons ship their own files in these folders. Rule files load in name order, so a later file wins for the same name.
; Rule forms: Name|pool sets or replaces a rule. Name|- removes it. Name|A,B uses several pools.
; Name|+C adds pool C to the rule that earlier files left (or creates it).
; Prints a WARN line when a folder has no entries, or a pool has no male or male_full names.
; Results are temp aux vars on the player:
;   *NotM:M:<mod index> = 1 for each allowed, loaded plugin
;   *NotM:R:<base name> = pool list, comma-separated
;   *NotM:W:<pool> = total names in the pool (weight for multi-pool rules)
;   *NotM:<pool>:male / :female / :surname / :male_full / :female_full = name arrays
ref rPlayer
ref rLoad
array_var aFiles
array_var aParts
array_var aPools
array_var aOld
string_var sDir
string_var sFile
string_var sLine
string_var sPool
string_var sSpec
int iCount
int iSize
int iFull
int iFem
int iFemFull
int iPart
int iFile
int iLine
int iModIdx
int iTotal
int iPlugins
array_var aRules
array_var aSeen

begin Function {}
	let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
	let rLoad := CompileScript "NamesOfTheMojave/LoadList.txt"
	let sDir := "Data/Config/NamesOfTheMojave/"
	let aRules := ar_Construct "stringmap"
	let aSeen := ar_Construct "stringmap"
	let iPlugins := 0

	let aFiles := GetFilesInFolder "Config/NamesOfTheMojave/Plugins" "*.txt"
	let iTotal := 0
	let iFile := 0
	while iFile < (ar_Size aFiles)
		let sFile := aFiles[iFile]
		let iCount := Call rLoad (sDir + "Plugins/" + sFile) "*NotM:Plugins"
		let iLine := 0
		while iLine < iCount
			let sLine := rPlayer.AuxVarGetStr "*NotM:Plugins" iLine
			let iModIdx := GetModIndex (sLine)
			if iModIdx < 255
				rPlayer.AuxVarSetFlt (sv_Construct "*NotM:M:%g" iModIdx) 1
				let iPlugins += 1
			endif
			let iLine += 1
		loop
		let iTotal += iCount
		let iFile += 1
	loop
	if iTotal == 0
		PrintToConsole "[NotM] WARN Plugins/ has no plugin names, no actor will be named"
	endif

	; Folder listing order is not guaranteed (MO2 merges folders), so sort for a fixed load order.
	let aFiles := GetFilesInFolder "Config/NamesOfTheMojave/Rules" "*.txt"
	if eval (ar_Size aFiles) > 1
		let aFiles := ar_Sort aFiles
	endif
	let iTotal := 0
	let iFile := 0
	while iFile < (ar_Size aFiles)
		let sFile := aFiles[iFile]
		let iCount := Call rLoad (sDir + "Rules/" + sFile) "*NotM:Rules"
		let iLine := 0
		while iLine < iCount
			let sLine := rPlayer.AuxVarGetStr "*NotM:Rules" iLine
			let aParts := sv_Split (sLine) "|"
			if eval (ar_Size aParts) != 2
				PrintToConsole "[NotM] WARN bad rule line [%z] in %z" sLine sFile
			elseif eval aParts[1] == "-"
				rPlayer.AuxVarErase ("*NotM:R:" + aParts[0])
				ar_Erase aRules (aParts[0])
			elseif eval aParts[1] == "+" || (sv_Length aParts[0]) == 0 || (sv_Length aParts[1]) == 0
				PrintToConsole "[NotM] WARN bad rule line [%z] in %z" sLine sFile
			else
				let sPool := aParts[1]
				let sSpec := sPool
				if eval sPool[0:0] == "+"
					; Append only pools the rule does not have yet: a duplicate would double its weight.
					let sPool := sPool[1:-1]
					let sSpec := rPlayer.AuxVarGetStr ("*NotM:R:" + aParts[0])
					let aOld := sv_Split (sSpec) ","
					let aPools := sv_Split (sPool) ","
					let iPart := 0
					while iPart < (ar_Size aPools)
						if eval (ar_Find aPools[iPart] aOld) == ar_BadNumericIndex
							if eval (sv_Length sSpec) > 0
								let sSpec := sSpec + "," + aPools[iPart]
							else
								let sSpec := aPools[iPart]
							endif
						endif
						let iPart += 1
					loop
				endif
				rPlayer.AuxVarSetStr ("*NotM:R:" + aParts[0]) (sSpec)
				let aRules[aParts[0]] := 1
				let aPools := sv_Split (sPool) ","
				let iPart := 0
				while iPart < (ar_Size aPools)
					let sPool := aPools[iPart]
					; An empty item from a stray comma (A,,B) is skipped.
					if eval (sv_Length sPool) > 0
						let aSeen[sPool] := 1
						if eval (rPlayer.AuxVarGetFlt ("*NotM:Loaded:" + sPool)) == 0
							let iSize := Call rLoad (sDir + "Names/" + sPool + "/male.txt") ("*NotM:" + sPool + ":male")
							let iFull := Call rLoad (sDir + "Names/" + sPool + "/male_full.txt") ("*NotM:" + sPool + ":male_full")
							if eval (iSize + iFull) == 0
								PrintToConsole "[NotM] WARN Names/%z has no male.txt or male_full.txt names" sPool
							endif
							let iFem := Call rLoad (sDir + "Names/" + sPool + "/female.txt") ("*NotM:" + sPool + ":female")
							let iFemFull := Call rLoad (sDir + "Names/" + sPool + "/female_full.txt") ("*NotM:" + sPool + ":female_full")
							Call rLoad (sDir + "Names/" + sPool + "/surname.txt") ("*NotM:" + sPool + ":surname")
							rPlayer.AuxVarSetFlt ("*NotM:W:" + sPool) (iSize + iFull + iFem + iFemFull)
							rPlayer.AuxVarSetFlt ("*NotM:Loaded:" + sPool) 1
						endif
					endif
					let iPart += 1
				loop
			endif
			let iLine += 1
		loop
		let iTotal += iCount
		let iFile += 1
	loop
	if iTotal == 0
		PrintToConsole "[NotM] WARN Rules/ has no rules, no actor will be named"
	endif
	let sLine := rPlayer.AuxVarGetStr "*NotM:Version"
	let iCount := ar_Size aRules
	let iTotal := ar_Size aSeen
	PrintToConsole "[NotM] Names of the Mojave %z: %g rules, %g pools, %g plugins" sLine iCount iTotal iPlugins
end
