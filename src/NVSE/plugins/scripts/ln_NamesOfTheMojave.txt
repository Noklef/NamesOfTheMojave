; Names of the Mojave loader. Runs on game load and new game.
; Compiles Main.gek and calls it 3 s later: actors are loaded and console output is not lost.
ref rMain
let rMain := CompileScript "NamesOfTheMojave/Main.gek"
if rMain
	CallAfter 3 rMain
else
	PrintToConsole "[NotM] ERROR Main.gek did not compile, see nvse.log"
endif
