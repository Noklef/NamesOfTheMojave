; Main-loop callback for Spike 3. Counts ticks in a temp aux var; stops itself at tick 3.
float fT
ref rSelf

begin Function {}
	let fT := PLAYER.AuxVarGetFlt "*NotMTick"
	let fT += 1
	PLAYER.AuxVarSetFlt "*NotMTick" fT
	if fT < 3
		PrintToConsole "NotM Spike3: INFO tick %g" fT
	elseif fT == 3
		PrintToConsole "NotM Spike3: PASS tick 3 reached"
		let rSelf := PLAYER.AuxVarGetRef "*NotMCB"
		SetGameMainLoopCallback rSelf 0
	endif
end
