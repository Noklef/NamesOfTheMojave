; Spike 1d: numeric local ref index for the name hash (GetLocalRefIndex).
; Runs 3 s after load: console prints during the load screen are lost.
CallAfter 3 (begin function {}
	ref rPlayer
	ref rActor
	int iLocal
	int iCount

	PrintToConsole "NotM Spike1d: START"
	let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
	let rActor := GetFirstRef 200 1
	while rActor
		if rActor != rPlayer
			let iLocal := GetLocalRefIndex rActor
			PrintToConsole "NotM Spike1d: INFO ref %i local %g" rActor iLocal
			let iCount += 1
		endif
		if iCount >= 8
			break
		endif
		let rActor := GetNextRef
	loop
	PrintToConsole "NotM Spike1d: END"
end)
