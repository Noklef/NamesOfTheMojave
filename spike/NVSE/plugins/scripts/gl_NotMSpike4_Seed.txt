; Spike 4: persistent JIP aux var on player (playthrough seed).
; PASS if after save + reload it prints FOUND with the same value.
; Tests a private name and a public ("_") name.
ref rPlayer
float fSeed

let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"

let fSeed := rPlayer.AuxVarGetFlt "NotMSpikeSeed"
if fSeed == 0
	let fSeed := Rand 1 1000000
	rPlayer.AuxVarSetFlt "NotMSpikeSeed" fSeed
	PrintToConsole "NotM Spike4: private seed CREATED %.0f" fSeed
else
	PrintToConsole "NotM Spike4: private seed FOUND %.0f" fSeed
endif

let fSeed := rPlayer.AuxVarGetFlt "_NotMSpikeSeed"
if fSeed == 0
	let fSeed := Rand 1 1000000
	rPlayer.AuxVarSetFlt "_NotMSpikeSeed" fSeed
	PrintToConsole "NotM Spike4: public seed CREATED %.0f" fSeed
else
	PrintToConsole "NotM Spike4: public seed FOUND %.0f" fSeed
endif
