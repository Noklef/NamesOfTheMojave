; Spike 1c: GetLeveledActorBase on nearby actors (real NPC base for leveled spawns?).
ref rPlayer
ref rActor
ref rBase
int iCount

let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
let rActor := GetFirstRef 200 1
while rActor
	if rActor != rPlayer
		let rBase := rActor.GetLeveledActorBase
		PrintToConsole "NotM Spike1c: ref %i lab %i name %n" rActor rBase rBase
		let iCount += 1
	endif
	if iCount >= 8
		break
	endif
	let rActor := GetNextRef
loop
