; Spike 1e: ref ID as hex string (GetFormIDString).
ref rPlayer
ref rActor
string_var sID
int iCount

let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
let rActor := GetFirstRef 200 1
while rActor
	if rActor != rPlayer
		let sID := GetFormIDString rActor
		PrintToConsole "NotM Spike1e: ref %i idstr %z" rActor sID
		let iCount += 1
	endif
	if iCount >= 8
		break
	endif
	let rActor := GetNextRef
loop
