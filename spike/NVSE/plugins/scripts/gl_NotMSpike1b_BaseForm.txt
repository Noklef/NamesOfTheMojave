; Spike 1b: GetBaseForm on nearby actors (compare with Spike 1 gbo).
; Runs 3 s after load: console prints during the load screen are lost.
CallAfter 3 (begin function {}
	ref rPlayer
	ref rActor
	ref rBase
	int iCount

	PrintToConsole "NotM Spike1b: START"
	let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
	let rActor := GetFirstRef 200 1
	while rActor
		if rActor != rPlayer
			let rBase := rActor.GetBaseForm
			PrintToConsole "NotM Spike1b: INFO ref %i gbf %i name %n" rActor rBase rBase
			let iCount += 1
		endif
		if iCount >= 8
			break
		endif
		let rActor := GetNextRef
	loop
	PrintToConsole "NotM Spike1b: END"
end)
