; Spike 3: lambda + SetGameMainLoopCallback.
; PASS prints at tick 3 (about 2 s apart, GameMode only), then the callback stops.
; Unregisters the previous load's callback; prints stop after 3 even if unregister fails.
ref rPlayer
ref rCB
ref rOld
float fTick

PrintToConsole "NotM Spike3: START"
let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
let rOld := rPlayer.AuxVarGetRef "*NotMCB"
if rOld
	SetGameMainLoopCallback rOld 0
endif
rPlayer.AuxVarSetFlt "*NotMTick" 0
let rCB := (begin function {}
	let fTick := rPlayer.AuxVarGetFlt "*NotMTick"
	let fTick += 1
	rPlayer.AuxVarSetFlt "*NotMTick" fTick
	if fTick < 3
		PrintToConsole "NotM Spike3: INFO tick %g" fTick
	elseif fTick == 3
		PrintToConsole "NotM Spike3: PASS tick 3 reached"
		SetGameMainLoopCallback rCB 0
	endif
end)
rPlayer.AuxVarSetRef "*NotMCB" rCB
SetGameMainLoopCallback rCB 1 120 1
PrintToConsole "NotM Spike3: INFO callback registered; close console and wait"
PrintToConsole "NotM Spike3: END"
