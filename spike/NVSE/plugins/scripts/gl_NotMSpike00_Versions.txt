int iMaj
int iRev
int iBeta
float fV

PrintToConsole "NotM Spike00: INFO JIP LN %g" fV
let fV := GetPluginVersion "JohnnyGuitarNVSE"
PrintToConsole "NotM Spike00: INFO JohnnyGuitar %g" fV
let fV := GetPluginVersion "ShowOffNVSE"
