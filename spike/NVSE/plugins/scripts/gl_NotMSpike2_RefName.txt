; Spike 2: SetRefName with literal names, alternating Alpha/Beta.
; PASS if actors in-game show different names (not all the same).
ref rPlayer
ref rActor
int iCount
int iMod

let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
let rActor := GetFirstRef 200 1
while rActor
	if rActor != rPlayer
		let iMod := iCount % 2
		if iMod == 0
			rActor.SetRefName "NotM Alpha"
		else
			rActor.SetRefName "NotM Beta"
		endif
		let iCount += 1
	endif
	let rActor := GetNextRef
loop
PrintToConsole "NotM Spike2: renamed %g actors Alpha/Beta" iCount
