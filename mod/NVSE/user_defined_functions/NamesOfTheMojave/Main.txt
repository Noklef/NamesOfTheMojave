; Entry point. Creates the playthrough seed if missing, loads name pools, then names nearby actors.
; Seed: private permanent aux var on the player, saved in the JIP co-save.
; Compiled UDFs and pools: temp aux vars on the player, rebuilt on every load.
ref rPlayer
ref rLoad
ref rScan
float fSeed
string_var sDir

begin Function {}
	let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
	let fSeed := rPlayer.AuxVarGetFlt "NotMSeed"
	if fSeed == 0
		let fSeed := floor (Rand 1 1000000)
		rPlayer.AuxVarSetFlt "NotMSeed" fSeed
	endif
	rPlayer.AuxVarSetRef "*NotM:Hash" (CompileScript "NamesOfTheMojave/Hash.txt")
	rPlayer.AuxVarSetRef "*NotM:Pick" (CompileScript "NamesOfTheMojave/PickName.txt")
	let rLoad := CompileScript "NamesOfTheMojave/LoadList.txt"
	let sDir := "Data/Config/NamesOfTheMojave/Names/wasteland/"
	Call rLoad (sDir + "male.txt") "*NotM:wasteland:male"
	Call rLoad (sDir + "surname.txt") "*NotM:wasteland:surname"
	let rScan := CompileScript "NamesOfTheMojave/Scan.txt"
	Call rScan
end
