; Names generic actors near the player.
; Vertical slice: Powder Gangers only, wasteland pool.
ref rPlayer
ref rActor
ref rPick
int iCount
string_var sBase
string_var sName

begin Function {}
	let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
	let rPick := rPlayer.AuxVarGetRef "*NotM:Pick"
	let rActor := GetFirstRef 200 1
	while rActor
		if rActor != rPlayer
			let sBase := rActor.GetName
			if eval sBase == "Powder Ganger"
				let sName := Call rPick rActor "wasteland" 0
				if eval sv_Length sName > 0
					rActor.SetRefName (sName)
					let iCount += 1
				endif
			endif
		endif
		let rActor := GetNextRef
	loop
	PrintToConsole "[NotM] named %g actors" iCount
end
