; Returns a name for an actor from a loaded pool.
; Gender: female lists if the actor is female and the pool has female names (female or female_full), else male lists.
; Attempts 0-3 pick an exact full name (<gender>_full). Later attempts, or a pool without full names,
; use a first name plus a surname if the pool has any. A pool without first names always uses full names.
; Same actor, seed and attempt always give the same name. A higher iAttempt re-rolls.
ref rActor
string_var sPool
int iAttempt
ref rPlayer
ref rHash
string_var sVar
string_var sName
string_var sLast
float fSeed
int iIdx
int iFull
int iSize
int iPick

begin Function {rActor, sPool, iAttempt}
	let rPlayer := GetFormFromMod "FalloutNV.esm" "000014"
	let fSeed := rPlayer.AuxVarGetFlt "NotMSeed"
	let rHash := rPlayer.AuxVarGetRef "*NotM:Hash"
	let iIdx := GetLocalRefIndex rActor
	let sVar := "*NotM:" + sPool + ":male"
	if eval rActor.GetIsSex Female
		if eval ((rPlayer.AuxVarSize ("*NotM:" + sPool + ":female")) + (rPlayer.AuxVarSize ("*NotM:" + sPool + ":female_full"))) > 0
			let sVar := "*NotM:" + sPool + ":female"
		endif
	endif
	let iFull := rPlayer.AuxVarSize (sVar + "_full")
	let iSize := rPlayer.AuxVarSize (sVar)
	if eval iFull > 0 && (iAttempt < 4 || iSize == 0)
		let iPick := Call rHash iIdx fSeed (iAttempt * 2 + 1) iFull
		let sName := rPlayer.AuxVarGetStr (sVar + "_full") iPick
		SetFunctionValue sName
		return
	endif
	if iSize == 0
		SetFunctionValue ""
		return
	endif
	let iPick := Call rHash iIdx fSeed (iAttempt * 2 + 1) iSize
	let sName := rPlayer.AuxVarGetStr (sVar) iPick
	let sVar := "*NotM:" + sPool + ":surname"
	let iSize := rPlayer.AuxVarSize (sVar)
	if iSize > 0
		let iPick := Call rHash iIdx fSeed (iAttempt * 2 + 2) iSize
		let sLast := rPlayer.AuxVarGetStr (sVar) iPick
		let sName := sName + " " + sLast
	endif
	SetFunctionValue sName
end
